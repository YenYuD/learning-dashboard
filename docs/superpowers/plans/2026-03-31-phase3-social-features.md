```typescript
import { router, protectedProcedure } from '../trpc';
import { z } from 'zod';
import { prisma } from '~/server/prisma';
import { TRPCError } from '@trpc/server';

const inviteRouter = router({
  create: protectedProcedure.mutation(async ({ ctx }) => {
    const invite = await prisma.friendInvite.create({
      data: {
        inviterId: ctx.userId,
        expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
      },
    });
    return { token: invite.token };
  }),

  validate: protectedProcedure
    .input(z.object({ token: z.string() }))
    .query(async ({ ctx, input }) => {
      const invite = await prisma.friendInvite.findUnique({
        where: { token: input.token },
        include: { inviter: { select: { name: true, image: true } } },
      });

      if (!invite) {
        return { valid: false as const, reason: 'not_found' as const };
      }
      if (invite.usedById) {
        return { valid: false as const, reason: 'used' as const };
      }
      if (invite.expiresAt < new Date()) {
        return { valid: false as const, reason: 'expired' as const };
      }
      if (invite.inviterId === ctx.userId) {
        return { valid: false as const, reason: 'self' as const };
      }

      // Check if already friends
      const existing = await prisma.friendship.findFirst({
        where: {
          OR: [
            { requesterId: invite.inviterId, addresseeId: ctx.userId },
            { requesterId: ctx.userId, addresseeId: invite.inviterId },
          ],
          status: 'ACCEPTED',
        },
      });
      if (existing) {
        return { valid: false as const, reason: 'already_friends' as const };
      }

      return {
        valid: true as const,
        inviterName: invite.inviter.name,
        inviterImage: invite.inviter.image,
      };
    }),

  use: protectedProcedure
    .input(z.object({
      token: z.string(),
      action: z.enum(['accept', 'decline']),
    }))
    .mutation(async ({ ctx, input }) => {
      const invite = await prisma.friendInvite.findUnique({
        where: { token: input.token },
      });

      if (!invite) throw new TRPCError({ code: 'NOT_FOUND', message: 'Invite not found' });
      if (invite.usedById) throw new TRPCError({ code: 'BAD_REQUEST', message: 'Invite already used' });
      if (invite.expiresAt < new Date()) throw new TRPCError({ code: 'BAD_REQUEST', message: 'Invite expired' });
      if (invite.inviterId === ctx.userId) throw new TRPCError({ code: 'BAD_REQUEST', message: 'Cannot invite yourself' });

      const status = input.action === 'accept' ? 'ACCEPTED' : 'DECLINED';

      await prisma.$transaction([
        prisma.friendship.create({
          data: {
            requesterId: invite.inviterId,
            addresseeId: ctx.userId,
            status: status as 'ACCEPTED' | 'DECLINED',
          },
        }),
        prisma.friendInvite.update({
          where: { id: invite.id },
          data: { usedById: ctx.userId, usedAt: new Date() },
        }),
      ]);

      return { status };
    }),
});

export const friendRouter = router({
  invite: inviteRouter,

  list: protectedProcedure.query(async ({ ctx }) => {
    const friendships = await prisma.friendship.findMany({
      where: {
        OR: [
          { requesterId: ctx.userId, status: 'ACCEPTED' },
          { addresseeId: ctx.userId, status: 'ACCEPTED' },
        ],
      },
      include: {
        requester: { select: { id: true, name: true, image: true } },
        addressee: { select: { id: true, name: true, image: true } },
      },
    });

    return friendships.map((f) => {
      const friend = f.requesterId === ctx.userId ? f.addressee : f.requester;
      return {
        friendshipId: f.id,
        id: friend.id,
        name: friend.name,
        image: friend.image,
        since: f.createdAt,
      };
    });
  }),

  pending: protectedProcedure.query(async ({ ctx }) => {
    return prisma.friendship.findMany({
      where: { addresseeId: ctx.userId, status: 'PENDING' },
      include: {
        requester: { select: { id: true, name: true, image: true } },
      },
      orderBy: { createdAt: 'desc' },
    });
  }),

  remove: protectedProcedure
    .input(z.object({ friendshipId: z.string() }))
    .mutation(async ({ ctx, input }) => {
      const friendship = await prisma.friendship.findUnique({
        where: { id: input.friendshipId },
      });

      if (!friendship) throw new TRPCError({ code: 'NOT_FOUND' });
      if (friendship.requesterId !== ctx.userId && friendship.addresseeId !== ctx.userId) {
        throw new TRPCError({ code: 'FORBIDDEN' });
      }

      await prisma.friendship.delete({ where: { id: input.friendshipId } });
      return { success: true };
    }),
});
```

```typescript
import { router, protectedProcedure } from '../trpc';
import { z } from 'zod';
import { prisma } from '~/server/prisma';

/**
 * Get all friend user IDs for a given user (including the user themselves)
 */
async function getFriendIdsIncludingSelf(userId: string): Promise<string[]> {
  const friendships = await prisma.friendship.findMany({
    where: {
      OR: [
        { requesterId: userId, status: 'ACCEPTED' },
        { addresseeId: userId, status: 'ACCEPTED' },
      ],
    },
    select: { requesterId: true, addresseeId: true },
  });

  const ids = new Set<string>([userId]);
  for (const f of friendships) {
    ids.add(f.requesterId);
    ids.add(f.addresseeId);
  }
  return [...ids];
}

/**
 * Calculate streak (consecutive days with at least one timeEntry) for a user
 */
async function calculateStreak(userId: string): Promise<number> {
  const entries = await prisma.timeEntry.findMany({
    where: { board: { user_id: userId } },
    select: { createdAt: true },
  });

  const dateSet = new Set(
    entries.map((e) => {
      const d = new Date(e.createdAt);
      return `${d.getFullYear()}-${d.getMonth()}-${d.getDate()}`;
    }),
  );

  let streak = 0;
  const cursor = new Date();
  cursor.setHours(0, 0, 0, 0);

  while (true) {
    const key = `${cursor.getFullYear()}-${cursor.getMonth()}-${cursor.getDate()}`;
    if (!dateSet.has(key)) break;
    streak++;
    cursor.setDate(cursor.getDate() - 1);
  }

  return streak;
}

/**
 * Get time range start date
 */
function getTimeRangeStart(timeRange: 'week' | 'month'): Date {
  const now = new Date();
  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());

  if (timeRange === 'week') {
    const dayOfWeek = today.getDay();
    const mondayOffset = (dayOfWeek + 6) % 7;
    today.setDate(today.getDate() - mondayOffset);
    return today;
  }

  // month
  return new Date(now.getFullYear(), now.getMonth(), 1);
}

export const rankingRouter = router({
  leaderboard: protectedProcedure
    .input(z.object({
      dimension: z.enum(['hours', 'streak', 'tasks']),
      timeRange: z.enum(['week', 'month']),
    }))
    .query(async ({ ctx, input }) => {
      const userIds = await getFriendIdsIncludingSelf(ctx.userId);

      const users = await prisma.user.findMany({
        where: { id: { in: userIds } },
        select: { id: true, name: true, image: true },
      });
      const userMap = new Map(users.map((u) => [u.id, u]));

      if (input.dimension === 'streak') {
        const streaks = await Promise.all(
          userIds.map(async (uid) => ({
            userId: uid,
            value: await calculateStreak(uid),
          })),
        );
        streaks.sort((a, b) => b.value - a.value);

        return streaks.map((s, i) => ({
          rank: i + 1,
          userId: s.userId,
          name: userMap.get(s.userId)?.name ?? null,
          image: userMap.get(s.userId)?.image ?? null,
          value: s.value,
        }));
      }

      const startDate = getTimeRangeStart(input.timeRange);

      if (input.dimension === 'hours') {
        const entries = await prisma.timeEntry.groupBy({
          by: ['boardId'],
          where: {
            board: { user_id: { in: userIds } },
            createdAt: { gte: startDate },
          },
          _sum: { duration: true },
        });

        // Map boardId -> userId
        const boards = await prisma.board.findMany({
          where: { id: { in: entries.map((e) => e.boardId) } },
          select: { id: true, user_id: true },
        });
        const boardOwner = new Map(boards.map((b) => [b.id, b.user_id]));

        // Aggregate per user
        const userTotals = new Map<string, number>();
        for (const uid of userIds) userTotals.set(uid, 0);

        for (const e of entries) {
          const uid = boardOwner.get(e.boardId);
          if (uid) {
            userTotals.set(uid, (userTotals.get(uid) ?? 0) + (e._sum.duration ?? 0));
          }
        }

        const sorted = [...userTotals.entries()].sort((a, b) => b[1] - a[1]);

        return sorted.map(([uid, value], i) => ({
          rank: i + 1,
          userId: uid,
          name: userMap.get(uid)?.name ?? null,
          image: userMap.get(uid)?.image ?? null,
          value,
        }));
      }

      // dimension === 'tasks'
      // Count tasks in "Done"/"Complete" lists
      const taskCounts = await Promise.all(
        userIds.map(async (uid) => {
          const count = await prisma.task.count({
            where: {
              list: {
                board: { user_id: uid },
                name: { in: ['Done', 'Complete', 'Completed'] },
              },
              createdAt: { gte: startDate },
            },
          });
          return { userId: uid, value: count };
        }),
      );

      taskCounts.sort((a, b) => b.value - a.value);

      return taskCounts.map((t, i) => ({
        rank: i + 1,
        userId: t.userId,
        name: userMap.get(t.userId)?.name ?? null,
        image: userMap.get(t.userId)?.image ?? null,
        value: t.value,
      }));
    }),

  myStats: protectedProcedure.query(async ({ ctx }) => {
    const weekStart = getTimeRangeStart('week');

    const [weeklyAgg, streak] = await Promise.all([
      prisma.timeEntry.aggregate({
        where: {
          board: { user_id: ctx.userId },
          createdAt: { gte: weekStart },
        },
        _sum: { duration: true },
      }),
      calculateStreak(ctx.userId),
    ]);

    const weeklyTasks = await prisma.task.count({
      where: {
        list: {
          board: { user_id: ctx.userId },
          name: { in: ['Done', 'Complete', 'Completed'] },
        },
        createdAt: { gte: weekStart },
      },
    });

    return {
      weeklyHours: weeklyAgg._sum.duration ?? 0,
      streak,
      weeklyTasks,
```

```typescript
import { router, protectedProcedure } from '../trpc';
import { z } from 'zod';
import { prisma } from '~/server/prisma';
import { TRPCError } from '@trpc/server';

/**
 * Verify the caller is an accepted friend of the target user.
 */
async function verifyFriendship(callerId: string, friendId: string) {
  const friendship = await prisma.friendship.findFirst({
    where: {
      OR: [
        { requesterId: callerId, addresseeId: friendId },
        { requesterId: friendId, addresseeId: callerId },
      ],
      status: 'ACCEPTED',
    },
  });
  if (!friendship) {
    throw new TRPCError({ code: 'FORBIDDEN', message: 'Not friends' });
  }
}

function getWeekStart(): Date {
  const now = new Date();
  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  const dayOfWeek = today.getDay();
  today.setDate(today.getDate() - ((dayOfWeek + 6) % 7));
  return today;
}

function calculateStreakFromDates(entries: { createdAt: Date }[]): number {
  const dateSet = new Set(
    entries.map((e) => {
      const d = new Date(e.createdAt);
      return `${d.getFullYear()}-${d.getMonth()}-${d.getDate()}`;
    }),
  );

  let streak = 0;
  const cursor = new Date();
  cursor.setHours(0, 0, 0, 0);

  while (true) {
    const key = `${cursor.getFullYear()}-${cursor.getMonth()}-${cursor.getDate()}`;
    if (!dateSet.has(key)) break;
    streak++;
    cursor.setDate(cursor.getDate() - 1);
  }
  return streak;
}

export const friendStatsRouter = router({
  getSummary: protectedProcedure
    .input(z.object({ friendId: z.string() }))
    .query(async ({ ctx, input }) => {
      await verifyFriendship(ctx.userId, input.friendId);

      const user = await prisma.user.findUnique({
        where: { id: input.friendId },
        select: { name: true, image: true },
      });

      const weekStart = getWeekStart();
      const userFilter = { board: { user_id: input.friendId } };

      const [weeklyAgg, weeklyTasks, allEntries] = await prisma.$transaction([
        prisma.timeEntry.aggregate({
          where: { ...userFilter, createdAt: { gte: weekStart } },
          _sum: { duration: true },
        }),
        prisma.task.count({
          where: {
            list: {
              board: { user_id: input.friendId },
              name: { in: ['Done', 'Complete', 'Completed'] },
            },
            createdAt: { gte: weekStart },
          },
        }),
        prisma.timeEntry.findMany({
          where: userFilter,
          select: { createdAt: true },
        }),
      ]);

      return {
        name: user?.name ?? null,
        image: user?.image ?? null,
        weeklyMinutes: weeklyAgg._sum.duration ?? 0,
        weeklyTasks,
        streak: calculateStreakFromDates(allEntries),
      };
    }),

  getWeeklyChart: protectedProcedure
    .input(z.object({ friendId: z.string() }))
    .query(async ({ ctx, input }) => {
      await verifyFriendship(ctx.userId, input.friendId);

      const weekStart = getWeekStart();
      const entries = await prisma.timeEntry.findMany({
        where: {
          board: { user_id: input.friendId },
          createdAt: { gte: weekStart },
        },
        select: { duration: true, createdAt: true },
      });

      const dayLabels = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
      const buckets = new Array(7).fill(0);

      for (const entry of entries) {
        const d = new Date(entry.createdAt);
        const dayIndex = (d.getDay() + 6) % 7;
        buckets[dayIndex] += entry.duration;
      }

      return dayLabels.map((day, i) => ({
        day,
        minutes: buckets[i],
      }));
    }),

  getBoardBreakdown: protectedProcedure
    .input(z.object({ friendId: z.string() }))
    .query(async ({ ctx, input }) => {
      await verifyFriendship(ctx.userId, input.friendId);

      const entries = await prisma.timeEntry.groupBy({
        by: ['boardId'],
        where: { board: { user_id: input.friendId } },
        _sum: { duration: true },
      });

      const boards = await prisma.board.findMany({
        where: { id: { in: entries.map((e) => e.boardId) } },
        select: { id: true, name: true, color: true },
      });
      const boardMap = new Map(boards.map((b) => [b.id, b]));

      return entries
        .map((e) => ({
          name: boardMap.get(e.boardId)?.name ?? 'Unknown',
          color: boardMap.get(e.boardId)?.color ?? null,
          minutes: e._sum.duration ?? 0,
        }))
        .sort((a, b) => b.minutes - a.minutes);
    }),
});
```

```typescript
import webpush from 'web-push';
import { prisma } from '~/server/prisma';

const VAPID_PUBLIC_KEY = process.env.VAPID_PUBLIC_KEY;
const VAPID_PRIVATE_KEY = process.env.VAPID_PRIVATE_KEY;
const VAPID_EMAIL = process.env.VAPID_EMAIL ?? 'mailto:noreply@example.com';

if (VAPID_PUBLIC_KEY && VAPID_PRIVATE_KEY) {
  webpush.setVapidDetails(VAPID_EMAIL, VAPID_PUBLIC_KEY, VAPID_PRIVATE_KEY);
}

export interface NotificationPayload {
  title: string;
  body: string;
  url?: string;
}

/**
 * Send push notification to a specific user (all their enabled subscriptions)
 */
export async function sendPushToUser(userId: string, payload: NotificationPayload) {
  if (!VAPID_PUBLIC_KEY || !VAPID_PRIVATE_KEY) return;

  const subscriptions = await prisma.pushSubscription.findMany({
    where: { userId, enabled: true },
  });

  const results = await Promise.allSettled(
    subscriptions.map((sub) =>
      webpush.sendNotification(
        {
          endpoint: sub.endpoint,
          keys: { p256dh: sub.p256dh, auth: sub.auth },
        },
        JSON.stringify(payload),
      ),
    ),
  );

  // Clean up invalid subscriptions (410 Gone or 404 Not Found)
  for (let i = 0; i < results.length; i++) {
    const result = results[i];
    if (result.status === 'rejected') {
      const statusCode = (result.reason as { statusCode?: number })?.statusCode;
      if (statusCode === 404 || statusCode === 410) {
        await prisma.pushSubscription.delete({ where: { id: subscriptions[i].id } });
      }
    }
  }
}

/**
 * Send push notification to multiple users
 */
export async function sendPushToUsers(userIds: string[], payload: NotificationPayload) {
  await Promise.allSettled(
    userIds.map((uid) => sendPushToUser(uid, payload)),
  );
}

/**
 * Get friend user IDs for a given user
 */
export async function getFriendIds(userId: string): Promise<string[]> {
  const friendships = await prisma.friendship.findMany({
    where: {
      OR: [
        { requesterId: userId, status: 'ACCEPTED' },
        { addresseeId: userId, status: 'ACCEPTED' },
      ],
    },
    select: { requesterId: true, addresseeId: true },
  });

  return friendships.map((f) =>
    f.requesterId === userId ? f.addresseeId : f.requesterId,
  );
}

/** Milestone thresholds in minutes */
export const MILESTONE_THRESHOLDS = [300, 600, 1200, 3000]; // 5hr, 10hr, 20hr, 50hr

/**
 * Check if a user just crossed a milestone and notify friends if so.
 * Call this after creating a timeEntry.
 */
export async function checkMilestoneAndNotify(userId: string, userName: string | null) {
  const weekStart = getWeekStart();

  const agg = await prisma.timeEntry.aggregate({
    where: {
      board: { user_id: userId },
      createdAt: { gte: weekStart },
    },
    _sum: { duration: true },
  });

  const totalMinutes = agg._sum.duration ?? 0;

  for (const threshold of MILESTONE_THRESHOLDS) {
    if (totalMinutes >= threshold && totalMinutes - threshold < 60) {
      // Just crossed this threshold (within the last 60 min of entries)
      const hours = threshold / 60;
      const friendIds = await getFriendIds(userId);

      await sendPushToUsers(friendIds, {
        title: 'Friend Milestone!',
        body: `${userName ?? 'A friend'} just hit ${hours} hours this week!`,
        url: `/friends/${userId}`,
      });
      break; // Only notify for the highest milestone crossed
    }
  }
}

function getWeekStart(): Date {
  const now = new Date();
  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  const dayOfWeek = today.getDay();
  today.setDate(today.getDate() - ((dayOfWeek + 6) % 7));
  return today;
}
```

```typescript
  if (friendIds.length === 0) return;

  const allIds = [userId, ...friendIds];

  weekStart.setDate(weekStart.getDate() - ((weekStart.getDay() + 6) % 7));

    select: { id: true, user_id: true },
  });
  const boardOwner = new Map(boards.map((b) => [b.id, b.user_id]));

  const totals = new Map<string, number>();
  for (const id of allIds) totals.set(id, 0);
  for (const e of entries) {
    const uid = boardOwner.get(e.boardId);
    if (uid) totals.set(uid, (totals.get(uid) ?? 0) + (e._sum.duration ?? 0));
  }

  const sorted = [...totals.entries()].sort((a, b) => b[1] - a[1]);
  const myRank = sorted.findIndex(([uid]) => uid === userId) + 1;

  // If user just moved up, notify the person they passed
  if (myRank > 1) return; // Only notify when someone gets overtaken

  // Check who is now rank 2 (they got passed)
  const passedUserId = sorted[1]?.[0];
  if (passedUserId && passedUserId !== userId) {
    await sendPushToUser(passedUserId, {
      title: 'Ranking Update',
      body: `${userName ?? 'Someone'} passed you! You're now ranked #2 this week.`,
      url: '/ranking',
    });
  }
}
```

```typescript
import { NextResponse } from 'next/server';
import { prisma } from '~/server/prisma';
import { sendPushToUser } from '~/server/routers/notification.service';

/**
 * Daily reminder cron endpoint.
 * Called by Google Cloud Scheduler at 20:00 Taiwan time (12:00 UTC).
 * Sends push notification to users who haven't studied today.
 */
export async function POST(request: Request) {
  // Verify request comes from Cloud Scheduler
  const authHeader = request.headers.get('authorization');
  const cronSecret = process.env.CRON_SECRET;

  if (cronSecret && authHeader !== `Bearer ${cronSecret}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const todayStart = new Date();
  todayStart.setHours(0, 0, 0, 0);

  // Find all users who have enabled push subscriptions
  const usersWithPush = await prisma.pushSubscription.findMany({
    where: { enabled: true },
    select: { userId: true },
    distinct: ['userId'],
  });

  const userIds = usersWithPush.map((u) => u.userId);
  if (userIds.length === 0) {
    return NextResponse.json({ sent: 0 });
  }

  // Find users who HAVE studied today
  const activeToday = await prisma.timeEntry.findMany({
    where: {
      board: { user_id: { in: userIds } },
      createdAt: { gte: todayStart },
    },
    select: { board: { select: { user_id: true } } },
    distinct: ['boardId'],
  });

  const activeUserIds = new Set(activeToday.map((e) => e.board.user_id));

  // Users who haven't studied today
  const inactiveUserIds = userIds.filter((uid) => !activeUserIds.has(uid));

  // Send reminders
  await Promise.allSettled(
    inactiveUserIds.map((uid) =>
      sendPushToUser(uid, {
        title: 'Study Reminder',
        body: "You haven't started studying today! Keep your streak going!",
        url: '/dashboard',
      }),
    ),
  );

  return NextResponse.json({ sent: inactiveUserIds.length });
}
```

```typescript
'use client';

import { useCallback, useEffect, useState } from 'react';
import { trpc } from '~/utils/trpc';

export function usePushSubscription() {
  const [permission, setPermission] = useState<NotificationPermission>('default');
  const subscribeMutation = trpc.notification.subscribe.useMutation();
  const unsubscribeMutation = trpc.notification.unsubscribe.useMutation();
  const toggleMutation = trpc.notification.toggle.useMutation();
  const statusQuery = trpc.notification.status.useQuery();

  useEffect(() => {
    if (typeof window !== 'undefined' && 'Notification' in window) {
      setPermission(Notification.permission);
    }
  }, []);

  const subscribe = useCallback(async () => {
    if (!('serviceWorker' in navigator) || !('PushManager' in window)) {
      return false;
    }

    const perm = await Notification.requestPermission();
    setPermission(perm);
    if (perm !== 'granted') return false;

    const registration = await navigator.serviceWorker.ready;
    const subscription = await registration.pushManager.subscribe({
      userVisibleOnly: true,
      applicationServerKey: process.env.NEXT_PUBLIC_VAPID_PUBLIC_KEY,
    });

    const json = subscription.toJSON();
    if (!json.endpoint || !json.keys?.p256dh || !json.keys?.auth) return false;

    await subscribeMutation.mutateAsync({
      endpoint: json.endpoint,
      p256dh: json.keys.p256dh,
      auth: json.keys.auth,
    });

    return true;
  }, [subscribeMutation]);

  const unsubscribe = useCallback(async () => {
    const registration = await navigator.serviceWorker.ready;
    const subscription = await registration.pushManager.getSubscription();
    if (subscription) {
      await unsubscribeMutation.mutateAsync({ endpoint: subscription.endpoint });
      await subscription.unsubscribe();
    }
  }, [unsubscribeMutation]);

  const toggle = useCallback(
    async (enabled: boolean) => {
      await toggleMutation.mutateAsync({ enabled });
      statusQuery.refetch();
    },
    [toggleMutation, statusQuery],
  );

  return {
    permission,
    isEnabled: statusQuery.data?.enabled ?? false,
    deviceCount: statusQuery.data?.deviceCount ?? 0,
    subscribe,
    unsubscribe,
    toggle,
  };
}
```

```tsx
'use client';

import { RankRow } from './RankRow';
import { useSession } from 'next-auth/react';

interface LeaderboardEntry {
  rank: number;
  userId: string;
  name: string | null;
  image: string | null;
  value: number;
}

interface LeaderboardTableProps {
  data: LeaderboardEntry[];
  dimension: 'hours' | 'streak' | 'tasks';
}

const UNITS: Record<string, string> = {
  hours: 'min',
  streak: 'days',
  tasks: 'tasks',
};

export function LeaderboardTable({ data, dimension }: LeaderboardTableProps) {
  const { data: session } = useSession();
  const myId = session?.user?.id;

  if (data.length === 0) {
    return (
      <p className="text-center text-gray-500 py-8">
        Add friends to see the leaderboard!
      </p>
    );
  }

  return (
    <div className="space-y-2">
      {data.map((entry) => (
        <RankRow
          key={entry.userId}
          rank={entry.rank}
          userId={entry.userId}
          name={entry.name}
          image={entry.image}
          value={entry.value}
          unit={UNITS[dimension]}
          isMe={entry.userId === myId}
        />
      ))}
    </div>
  );
}
```

```tsx
'use client';

import { useState } from 'react';
import { trpc } from '~/utils/trpc';
import { LeaderboardTable } from '~/components/ranking/LeaderboardTable';

type Dimension = 'hours' | 'streak' | 'tasks';
type TimeRange = 'week' | 'month';

export default function RankingPage() {
  const [dimension, setDimension] = useState<Dimension>('hours');
  const [timeRange, setTimeRange] = useState<TimeRange>('week');

  const leaderboardQuery = trpc.ranking.leaderboard.useQuery({
    dimension,
    timeRange,
  });

  const dimensions: { key: Dimension; label: string }[] = [
    { key: 'hours', label: 'Study Hours' },
    { key: 'streak', label: 'Streak' },
    { key: 'tasks', label: 'Tasks Completed' },
  ];

  return (
    <div className="mx-auto max-w-2xl space-y-6 p-6">
      <h1 className="text-2xl font-bold">Leaderboard</h1>

      {/* Dimension Tabs */}
      <div className="flex gap-1 rounded-lg bg-gray-100 p-1">
        {dimensions.map((d) => (
          <button
            key={d.key}
            onClick={() => setDimension(d.key)}
            className={`flex-1 rounded-md px-3 py-2 text-sm font-medium transition-colors ${
              dimension === d.key
                ? 'bg-white shadow-sm'
                : 'text-gray-600 hover:text-gray-900'
            }`}
          >
            {d.label}
          </button>
        ))}
      </div>

      {/* Time Range Toggle (not for streak) */}
      {dimension !== 'streak' && (
        <div className="flex gap-2">
          <button
            onClick={() => setTimeRange('week')}
            className={`rounded-full px-4 py-1.5 text-sm ${
              timeRange === 'week'
                ? 'bg-blue-100 text-blue-700'
                : 'text-gray-500 hover:bg-gray-100'
            }`}
          >
            This Week
          </button>
          <button
            onClick={() => setTimeRange('month')}
            className={`rounded-full px-4 py-1.5 text-sm ${
              timeRange === 'month'
                ? 'bg-blue-100 text-blue-700'
                : 'text-gray-500 hover:bg-gray-100'
            }`}
          >
            This Month
          </button>
        </div>
      )}

      {/* Leaderboard */}
      {leaderboardQuery.isLoading ? (
        <p className="text-gray-500">Loading...</p>
      ) : (
        <LeaderboardTable
          data={leaderboardQuery.data ?? []}
          dimension={dimension}
        />
      )}
    </div>
  );
}
```

```tsx
'use client';

import { usePushSubscription } from '~/hooks/usePushSubscription';

export function NotificationToggle() {
  const { permission, isEnabled, deviceCount, subscribe, toggle } = usePushSubscription();

  const handleToggle = async () => {
    if (permission === 'default' || permission === 'denied') {
      // Need to request permission first
      await subscribe();
    } else {
      await toggle(!isEnabled);
    }
  };

  return (
    <div className="flex items-center justify-between rounded-lg border p-4">
      <div>
        <p className="font-medium">Push Notifications</p>
        <p className="text-sm text-gray-500">
          {permission === 'denied'
            ? 'Notifications blocked by browser. Enable in browser settings.'
            : `${deviceCount} device(s) registered`}
        </p>
      </div>
      <button
        onClick={handleToggle}
        disabled={permission === 'denied'}
        className={`relative h-6 w-11 rounded-full transition-colors ${
          isEnabled ? 'bg-blue-500' : 'bg-gray-300'
        } ${permission === 'denied' ? 'opacity-50 cursor-not-allowed' : 'cursor-pointer'}`}
      >
        <span
          className={`absolute top-0.5 left-0.5 h-5 w-5 rounded-full bg-white transition-transform ${
            isEnabled ? 'translate-x-5' : ''
          }`}
        />
      </button>
    </div>
  );
}
```

This component can be added to an existing settings page or to the friends page. Integration into the app layout depends on whether a settings page already exists.
